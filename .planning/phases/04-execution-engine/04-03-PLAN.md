**class-result-page.php:**
```php
<?php
namespace TSM\Admin;

use TSM\Services\ExecutionService;
use TSM\Services\OutputFormatter;
use TSM\Services\ScriptService;
use TSM\Security;

class Result_Page {

    /**
     * Initialize the result page.
     * Registers the standalone page endpoint.
     */
    public static function init() {
        add_action( 'admin_init', array( __CLASS__, 'handle_result_page' ) );
    }

    /**
     * Handle the result page request.
     * Triggered via admin.php?page=tsm-result&execution_id=X
     */
    public static function handle_result_page() {
        if ( ! isset( $_GET['page'] ) || 'tsm-result' !== $_GET['page'] ) {
            return;
        }

        // Security check
        if ( ! Security::check_admin_permission() ) {
            wp_die( __( 'Unauthorized access.', 'test-script-manager' ) );
        }

        // Get execution ID
        $execution_id = isset( $_GET['execution_id'] ) ? absint( $_GET['execution_id'] ) : 0;
        if ( ! $execution_id ) {
            wp_die( __( 'Invalid execution ID.', 'test-script-manager' ) );
        }

        // Load execution data
        $execution = self::get_execution( $execution_id );
        if ( ! $execution ) {
            wp_die( __( 'Execution not found.', 'test-script-manager' ) );
        }

        // Get script info
        $script = ScriptService::get( $execution['script_id'] );

        // Enqueue assets
        self::enqueue_assets();

        // Render the page
        self::render( $execution, $script );
        exit;
    }

    /**
     * Get execution data from database.
     */
    private static function get_execution( $id ) {
        global $wpdb;

        $table = $wpdb->prefix . \TSM\Database::TABLE_EXECUTION_LOGS;
        $row = $wpdb->get_row(
            $wpdb->prepare( "SELECT * FROM $table WHERE id = %d", $id ),
            ARRAY_A
        );

        if ( $row ) {
            $row['error'] = json_decode( $row['error'], true );
        }

        return $row;
    }

    /**
     * Enqueue CSS and JS for result page.
     */
    private static function enqueue_assets() {
        wp_enqueue_style(
            'tsm-execution-result',
            TSM_PLUGIN_URL . 'assets/css/execution-result.css',
            array(),
            TSM_VERSION
        );

        wp_enqueue_script(
            'tsm-json-view',
            'https://cdn.jsdelivr.net/npm/json-view-for-chrome@latest/dist/json-view.min.js',
            array(),
            null,
            true
        );

        wp_enqueue_script(
            'tsm-execution-result',
            TSM_PLUGIN_URL . 'assets/js/execution-result.js',
            array( 'tsm-json-view' ),
            TSM_VERSION,
            true
        );
    }

    /**
     * Render the result page.
     */
    public static function render( $execution, $script ) {
        $script_name = $script ? $script['name'] : __( 'Unknown Script', 'test-script-manager' );

        // Prepare data for template
        $data = array(
            'script_name'    => $script_name,
            'status'         => $execution['status'],
            'output'         => $execution['output'],
            'errors'         => $execution['error']['errors'] ?? array(),
            'exception'      => $execution['error']['exception'] ?? null,
            'fatal'          => $execution['error']['fatal'] ?? null,
            'execution_time' => $execution['execution_time'],
            'memory_usage'   => $execution['memory_usage'],
            'executed_at'    => $execution['executed_at'],
        );

        // Detect output type for special formatting
        $data['output_type'] = OutputFormatter::detect_output_type( $execution['output'] );

        include TSM_PLUGIN_DIR . 'includes/admin/views/execution-result.php';
    }
}
```

```javascript
/**
 * Execution Result Page JavaScript
 *
 * Initializes JSON viewers for structured data display.
 */
(function() {
    'use strict';

    /**
     * Initialize JSON viewers on page load.
     */
    function initJsonViewers() {
        var viewers = document.querySelectorAll('.tsm-json-viewer');

        viewers.forEach(function(container) {
            var jsonStr = container.getAttribute('data-json');

            if (!jsonStr) {
                return;
            }

            try {
                var data = JSON.parse(jsonStr);

                // Check if json-view library is available
                if (typeof JSONView !== 'undefined') {
                    var view = new JSONView('json', data);
                    container.innerHTML = '';
                    container.appendChild(view.dom);

                    // Collapse by default for large objects
                    if (JSON.stringify(data).length > 1000) {
                        view.collapse(1);
                    }
                } else {
                    // Fallback: pretty print
                    var pre = document.createElement('pre');
                    pre.textContent = JSON.stringify(data, null, 2);
                    pre.style.color = '#d4d4d4';
                    pre.style.margin = '0';
                    container.appendChild(pre);
                }
            } catch (e) {
                // Not valid JSON, show as text
                var pre = document.createElement('pre');
                pre.textContent = jsonStr;
                pre.style.color = '#d4d4d4';
                pre.style.margin = '0';
                container.appendChild(pre);
            }
        });
    }

    /**
     * Copy output to clipboard.
     */
    function initCopyButton() {
        var copyBtn = document.querySelector('.tsm-copy-output');
        if (!copyBtn) return;

        copyBtn.addEventListener('click', function() {
            var output = document.querySelector('.tsm-output-content');
            if (!output) return;

            navigator.clipboard.writeText(output.textContent).then(function() {
                copyBtn.textContent = 'Copied!';
                setTimeout(function() {
                    copyBtn.textContent = 'Copy';
                }, 2000);
            });
        });
    }

    // Initialize on DOM ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            initJsonViewers();
            initCopyButton();
        });
    } else {
        initJsonViewers();
        initCopyButton();
    }
})();
```

```javascript
// Execute button handler
$('#tsm-execute-btn').on('click', function() {
    var scriptId = $('#tsm-current-script-id').val();
    if (!scriptId) {
        alert('Please save the script first.');
        return;
    }

    var btn = $(this);
    btn.prop('disabled', true).text('Executing...');

    $.ajax({
        url: tsmData.restUrl + 'scripts/' + scriptId + '/execute',
        method: 'POST',
        beforeSend: function(xhr) {
            xhr.setRequestHeader('X-WP-Nonce', tsmData.nonce);
        },
        data: {
            timeout: 30
        },
        success: function(response) {
            if (response.success) {
                // Get execution ID from the logged result
                // Open result page in new tab
                var resultUrl = tsmData.adminUrl + '?page=tsm-result&execution_id=' + response.result.execution_id;
                window.open(resultUrl, '_blank');
            } else {
                alert('Execution failed: ' + response.error);
            }
        },
        error: function(xhr) {
            var msg = xhr.responseJSON && xhr.responseJSON.error
                ? xhr.responseJSON.error
                : 'Execution failed';
            alert(msg);
        },
        complete: function() {
            btn.prop('disabled', false).text('Execute');
        }
    });
});
```
