Update `assets/js/admin-page.js`:
1. Add theme toggle function:
   ```javascript
   function initThemeToggle() {
     var toggleBtn = document.getElementById('tsm-theme-toggle');
     var themeLabel = document.getElementById('tsm-theme-label');
     if (!toggleBtn) return;

     // Get saved theme or default to vs-dark
     var currentTheme = localStorage.getItem('tsm-editor-theme') || 'vs-dark';
     updateThemeLabel(currentTheme);

     toggleBtn.addEventListener('click', function() {
       var newTheme = currentTheme === 'vs-dark' ? 'vs' : 'vs-dark';

       // Apply to all Monaco editors
       monaco.editor.setTheme(newTheme);

       // Save preference
       localStorage.setItem('tsm-editor-theme', newTheme);
       currentTheme = newTheme;

       updateThemeLabel(newTheme);
     });
   }

   function updateThemeLabel(theme) {
     var themeLabel = document.getElementById('tsm-theme-label');
     if (themeLabel) {
       themeLabel.textContent = theme === 'vs-dark' ? 'Light Theme' : 'Dark Theme';
     }
   }
   ```

2. Use saved theme when creating editor:
   ```javascript
   var savedTheme = localStorage.getItem('tsm-editor-theme') || 'vs-dark';

   var editor = monaco.editor.create(container, {
     // ... other options
     theme: savedTheme
   });
   ```

3. Initialize theme toggle after Monaco loads:
   ```javascript
   // In Monaco ready callback
   initThemeToggle();
   ```

Available Monaco themes:
- 'vs' - Light theme
- 'vs-dark' - Dark theme (default)
- 'hc-black' - High contrast (optional, for accessibility)
