function renderExecutionItem(execution) {
    const modeIcon = execution.execution_mode === 'background'
        ? '<span class="dashicons dashicons-cloud tsm-execution-mode-icon background" title="Background execution"></span>'
        : '<span class="dashicons dashicons-flash tsm-execution-mode-icon sync" title="Sync execution"></span>';

    // Status badge with retry count
    let statusText = execution.status;
    if (execution.status === 'retry' && execution.retry_count > 0) {
        statusText = `Retry ${execution.retry_count}/3`;
    }
    const statusBadge = `<span class="tsm-status-badge ${execution.status}">${statusText}</span>`;

    // Cancel button for cancellable states
    const isCancellable = ['pending', 'running', 'retry'].includes(execution.status);
    const cancelBtn = isCancellable && execution.execution_mode === 'background'
        ? `<button type="button" class="tsm-cancel-btn" data-execution-id="${execution.id}" title="Cancel execution">&times;</button>`
        : '';

    return `
        <div class="tsm-execution-item" data-execution-id="${execution.id}">
            <div class="tsm-execution-item-left">
                ${modeIcon}
                <span class="tsm-execution-time">${formatTime(execution.executed_at)}</span>
            </div>
            <div class="tsm-execution-item-right">
                ${statusBadge}
                ${cancelBtn}
            </div>
        </div>
    `;
}
```

/**
 * Execute script in background.
 */
async function executeBackground() {
    const scriptId = getCurrentScriptId();
    if (!scriptId) {
        alert(tsmL10n.selectScriptFirst || 'Please select a script first.');
        return;
    }

    // Auto-save before execute (same as sync execute)
    if (typeof saveScript === 'function') {
        await saveScript();
    }

    const btn = document.getElementById('tsm-execute-background-btn');
    btn.disabled = true;
    btn.innerHTML = '<span class="dashicons dashicons-update spin"></span> ' + (tsmL10n.scheduling || 'Scheduling...');

    try {
        const response = await fetch(`${tsmApiBase}/scripts/${scriptId}/execute-background`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-WP-Nonce': tsmNonce
            }
        });

        const data = await response.json();

        if (data.success) {
            // Show success message
            showNotice('success', tsmL10n.backgroundScheduled || 'Script scheduled for background execution. You will be notified when complete.');

            // Refresh execution history
            if (typeof loadExecutionHistory === 'function') {
                loadExecutionHistory(scriptId);
            }
        } else {
            showNotice('error', data.error || 'Failed to schedule background execution.');
        }
    } catch (error) {
        console.error('Background execute error:', error);
        showNotice('error', 'Network error. Please try again.');
    } finally {
        btn.disabled = false;
        btn.innerHTML = '<span class="dashicons dashicons-cloud"></span> ' + (tsmL10n.background || 'Background');
    }
}

/**
 * Cancel execution with confirmation.
 */
async function cancelExecution(executionId) {
    // Per CONTEXT.md: confirmation dialog required
    const confirmed = confirm(tsmL10n.confirmCancel || 'Are you sure you want to cancel this execution?');
    if (!confirmed) {
        return;
    }

    try {
        const response = await fetch(`${tsmApiBase}/executions/${executionId}/cancel`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-WP-Nonce': tsmNonce
            }
        });

        const data = await response.json();

        if (data.success) {
            showNotice('success', tsmL10n.executionCancelled || 'Execution cancelled.');

            // Update the item in history without full reload
            const item = document.querySelector(`.tsm-execution-item[data-execution-id="${executionId}"]`);
            if (item) {
                const badge = item.querySelector('.tsm-status-badge');
                if (badge) {
                    badge.className = 'tsm-status-badge cancelled';
                    badge.textContent = 'Cancelled';
                }
                const cancelBtn = item.querySelector('.tsm-cancel-btn');
                if (cancelBtn) {
                    cancelBtn.remove();
                }
            }
        } else {
            showNotice('error', data.error || 'Failed to cancel execution.');
        }
    } catch (error) {
        console.error('Cancel execution error:', error);
        showNotice('error', 'Network error. Please try again.');
    }
}

/**
 * Show inline notice.
 */
function showNotice(type, message) {
    // Check if there's an existing notice container
    let noticeContainer = document.getElementById('tsm-notices');
    if (!noticeContainer) {
        // Create notice container at top of content area
        noticeContainer = document.createElement('div');
        noticeContainer.id = 'tsm-notices';
        const contentArea = document.querySelector('.tsm-editor-area') || document.querySelector('.wrap');
        if (contentArea) {
            contentArea.insertBefore(noticeContainer, contentArea.firstChild);
        }
    }

    const notice = document.createElement('div');
    notice.className = `notice notice-${type} is-dismissible`;
    notice.innerHTML = `<p>${message}</p><button type="button" class="notice-dismiss"><span class="screen-reader-text">Dismiss</span></button>`;

    noticeContainer.appendChild(notice);

    // Auto-dismiss after 5 seconds
    setTimeout(() => {
        notice.style.transition = 'opacity 0.3s';
        notice.style.opacity = '0';
        setTimeout(() => notice.remove(), 300);
    }, 5000);

    // Manual dismiss
    notice.querySelector('.notice-dismiss').addEventListener('click', () => {
        notice.remove();
    });
}

// Event listeners
document.addEventListener('DOMContentLoaded', function() {
    // Background execute button
    const bgExecBtn = document.getElementById('tsm-execute-background-btn');
    if (bgExecBtn) {
        bgExecBtn.addEventListener('click', executeBackground);
    }

    // Cancel buttons (event delegation for dynamically added buttons)
    document.addEventListener('click', function(e) {
        if (e.target.classList.contains('tsm-cancel-btn')) {
            const executionId = e.target.dataset.executionId;
            if (executionId) {
                cancelExecution(executionId);
            }
        }
    });
});
```
