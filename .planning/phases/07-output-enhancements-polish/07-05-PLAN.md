2. **Add bulk_action() callback method**:
   ```php
   public function bulk_action(WP_REST_Request $request) {
       $action = $request->get_param('action');
       $script_ids = $request->get_param('script_ids');
       $results = array('success' => 0, 'failed' => 0, 'errors' => array());

       if (empty($script_ids)) {
           return new WP_Error('tsm_no_scripts', 'No scripts selected', array('status' => 400));
       }

       switch ($action) {
           case 'delete':
               foreach ($script_ids as $id) {
                   $result = ScriptService::delete($id);
                   if (is_wp_error($result)) {
                       $results['failed']++;
                       $results['errors'][] = $id . ': ' . $result->get_error_message();
                   } else {
                       $results['success']++;
                   }
               }
               break;

           case 'set_category':
               $category_id = $request->get_param('category_id');
               if (!$category_id) {
                   return new WP_Error('tsm_no_category', 'Category ID required', array('status' => 400));
               }
               foreach ($script_ids as $id) {
                   $result = CategoryService::set_script_categories($id, array($category_id));
                   if (is_wp_error($result)) {
                       $results['failed']++;
                       $results['errors'][] = $id . ': ' . $result->get_error_message();
                   } else {
                       $results['success']++;
                   }
               }
               break;
       }

       return rest_ensure_response($results);
   }
   ```

   ```javascript
   // Toggle bulk mode
   function toggleBulkMode() {
       const isActive = document.querySelector('.tsm-bulk-actions').style.display !== 'none';
       document.querySelector('.tsm-bulk-actions').style.display = isActive ? 'none' : 'flex';
       document.querySelectorAll('.tsm-script-checkbox').forEach(cb => {
           cb.style.display = isActive ? 'none' : 'inline';
           cb.checked = false;
       });
       updateSelectedCount();
   }

   // Select all toggle
   document.getElementById('tsm-select-all').addEventListener('change', function() {
       const checkboxes = document.querySelectorAll('.tsm-script-checkbox:not([style*="display: none"])');
       checkboxes.forEach(cb => cb.checked = this.checked);
       updateSelectedCount();
   });

   // Update selected count
   function updateSelectedCount() {
       const count = document.querySelectorAll('.tsm-script-checkbox:checked').length;
       document.querySelector('.tsm-selected-count').textContent = count + ' selected';
       document.querySelector('.tsm-apply-bulk').disabled = count === 0;
   }

   // Handle action change (show/hide category selector)
   document.getElementById('tsm-bulk-action-select').addEventListener('change', function() {
       const categorySelect = document.getElementById('tsm-bulk-category-select');
       categorySelect.style.display = this.value === 'set_category' ? 'inline' : 'none';
   });

   // Apply bulk action
   document.querySelector('.tsm-apply-bulk').addEventListener('click', async function() {
       const action = document.getElementById('tsm-bulk-action-select').value;
       if (!action) return;

       const scriptIds = Array.from(document.querySelectorAll('.tsm-script-checkbox:checked'))
           .map(cb => parseInt(cb.dataset.scriptId));

       if (scriptIds.length === 0) return;

       // Confirmation
       const confirmMsg = action === 'delete'
           ? `Delete ${scriptIds.length} scripts? This cannot be undone.`
           : `Set category for ${scriptIds.length} scripts?`;
       if (!confirm(confirmMsg)) return;

       const payload = { action, script_ids: scriptIds };
       if (action === 'set_category') {
           payload.category_id = parseInt(document.getElementById('tsm-bulk-category-select').value);
       }

       const response = await fetch('/wp-json/test-script-manager/v1/scripts/bulk', {
           method: 'POST',
           headers: {
               'Content-Type': 'application/json',
               'X-WP-Nonce': tsmData.nonce
           },
           body: JSON.stringify(payload)
       });

       const result = await response.json();
       alert(`Success: ${result.success}, Failed: ${result.failed}`);

       // Refresh script list
       if (action === 'delete') {
           scriptIds.forEach(id => {
               const row = document.querySelector(`[data-script-id="${id}"]`).closest('.tsm-script-item');
               if (row) row.remove();
           });
       }
       toggleBulkMode();
   });
   ```
